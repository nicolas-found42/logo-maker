"""Three preliminary flat arguments from the current frozen Rootshare brief; no selection."""

import json
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from logo_maker import briefing, motif
from logo_maker.paths import LogoDir


root = (Path(__file__).parent / "rootshare-v3").resolve()
frozen = briefing.current(LogoDir(root))
studies = root / "ideation/flat-studies"
studies.mkdir(parents=True, exist_ok=True)
scripts = {
    "shared-chamber": '''from pathlib import Path
from logo_maker.motif import Motif
m = Motif()
half = m.path("M490 150 C330 150 190 280 190 455 L190 610 C190 755 290 845 420 845 L420 700 C365 700 335 660 335 600 L335 455 C335 360 400 295 490 295 Z")
other = m.transform(half, rotate_deg=180, translate=(1000, 1000))
form = m.fit(m.union(half, other))
m.add("shared_chamber", form, fill="#B95E3F", role="primary")
m.save(Path(__file__).parent)
''',
    "seed-register": '''from pathlib import Path
from logo_maker.motif import Motif
m = Motif()
outer = m.path("M300 145 L700 145 C760 145 800 190 800 250 L800 745 C800 815 755 855 685 855 L315 855 C245 855 200 815 200 745 L200 250 C200 190 240 145 300 145 Z")
for y in (300, 500, 700):
    pocket = m.path(f"M310 {y} C375 {y-90} 560 {y-90} 690 {y} C560 {y+80} 375 {y+80} 310 {y} Z")
    outer = m.difference(outer, pocket)
m.add("seed_register", m.fit(outer), fill="#B95E3F", role="primary")
m.save(Path(__file__).parent)
''',
    "shared-initials": '''from pathlib import Path
from logo_maker.motif import Motif
m = Motif()
r, _ = m.glyphs("R", "Manrope:wght=700", 680, 190, 750)
s, _ = m.glyphs("S", "Manrope:wght=700", 680, 520, 750)
form = m.union(r, s)
bridge = m.rect(480, 440, 100, 90, rx=35)
form = m.union(form, bridge)
m.add("shared_initials", m.fit(form), fill="#B95E3F", role="primary")
m.save(Path(__file__).parent)
''',
}
hypotheses = {
    "shared-chamber": ("Complementary forms create one retained interior", "A resource held in common", "Generic chain-link or app icon without enough name specificity"),
    "seed-register": ("Three retained seed counters form a compact collection", "A maintained collection rather than one generic sprout", "Container or interface-list reading overwhelms seed-library context"),
    "shared-initials": ("R and S share a constructed contact", "Name-led identity, supported by the surrounding service name", "Stock initials or unclear ligature rather than a distinctive relationship"),
}
rows = []
for name, code in scripts.items():
    folder = studies / name
    folder.mkdir(exist_ok=True)
    (folder / "motif.py").write_text(code)
    svg = motif.run_motif_script(folder)
    errors = motif.validate(svg)
    if errors:
        raise RuntimeError((name, errors))
    motif.size_sheet(svg, folder / "sizes.png")
    motif.rasterize(svg, 512, folder / "flat.png", background="#F8F2E8")
    hypothesis, communication, risk = hypotheses[name]
    record = {"study": name, "hypothesis": hypothesis, "communication": communication,
              "risk": risk, "requirements_revision": frozen["requirements_revision"],
              "status": "preliminary_study", "owner_review": "unknown", "eligibility": "not_assessed"}
    (folder / "study.json").write_text(json.dumps(record, indent=2) + "\n")
    rows.append(record)
font_path = Path("/Users/nicolas/.cache/logo-maker/fonts/static/manrope-6f814d24ffa3.ttf")
font = ImageFont.truetype(str(font_path), 27)
small = ImageFont.truetype(str(font_path), 20)
sheet = Image.new("RGB", (1740, 760), "#F8F2E8")
draw = ImageDraw.Draw(sheet)
for i, row in enumerate(rows):
    x = i * 580
    with Image.open(studies / row["study"] / "flat.png") as frame:
        sheet.paste(frame.convert("RGB"), (x + 34, 70))
    draw.text((x + 34, 22), row["study"].replace("-", " ").title(), fill="#422D24", font=font)
    draw.text((x + 34, 600), "Rootshare / preliminary flat study", fill="#422D24", font=small)
    draw.text((x + 34, 636), "Craft and fit eligibility: not assessed", fill="#422D24", font=small)
    with Image.open(studies / row["study"] / "flat.png") as frame:
        pixel_view = frame.resize((32, 32), Image.Resampling.LANCZOS).resize((96, 96), Image.Resampling.NEAREST)
        sheet.paste(pixel_view.convert("RGB"), (x + 440, 640))
sheet.save(studies / "contact-sheet.png")
(studies / "studies.json").write_text(json.dumps(rows, indent=2) + "\n")
print(studies / "contact-sheet.png")
