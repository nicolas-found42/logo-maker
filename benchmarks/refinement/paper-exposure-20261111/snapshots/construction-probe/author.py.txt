def rounded_line(points,r=.10):

 return LineString([a,*[center+r*np.array([np.cos(t),np.sin(t)]) for t in np.linspace(t0,t0+delta,48)],c])

def along_y(poly,y0,y1):

def along_x(poly,x0,x1):

def slab(poly):
